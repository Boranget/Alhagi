## 菜单系统概述

### 菜单分类

本系统菜单系统分为三类：

1. **顶部菜单栏** - 主菜单，包含 6 个主菜单项
2. **上下文菜单（右键菜单）** - 根据点击位置动态显示
3. **编辑器斜杠菜单** - Milkdown 原生斜杠菜单

---

## 一、顶部菜单栏
标题栏存在 汉堡菜单按钮（☰），与菜单栏菜单项相同，使用electron自带菜单实现；建议菜单作为一个配置文件，一个配置多处使用

### 1.1 菜单结构

```
┌─────────────────────────────────────────────────────────────────┐
│  文件 (File)  编辑 (Edit)  视图 (View)  格式 (Format)  窗口 (Window)  帮助 (Help)  │
└─────────────────────────────────────────────────────────────────┘
```

## 三、菜单项状态逻辑

### 3.1 Enabled/Disabled 规则

```typescript
interface MenuItemState {
  enabled: boolean;   // 是否可点击
  visible: boolean;   // 是否显示
  checked?: boolean;  // 是否选中（用于开关型菜单）
}

// 菜单项状态计算函数
function calculateMenuItemState(menuId: string): MenuItemState {
  const state = {
    enabled: true,
    visible: true,
    checked: false
  };

  switch (menuId) {
    // 文件菜单
    case 'file.save':
    case 'file.saveAs':
    case 'file.rename':
    case 'file.copyPath':
      state.enabled = !!tabsStore.activeTab;
      break;

    case 'file.closeTab':
      state.enabled = tabsStore.tabs.size > 0;
      break;

    // 编辑菜单
    case 'edit.undo':
      state.enabled = editorStore.undoStack.length > 0;
      break;

    case 'edit.redo':
      state.enabled = editorStore.redoStack.length > 0;
      break;

    case 'edit.cut':
    case 'edit.copy':
    case 'edit.copyAsMarkdown':
    case 'edit.copyAsHtml':
      state.enabled = editorStore.hasSelection;
      break;

    // 视图菜单 - 开关型菜单
    case 'view.toggleTypewriterMode':
      state.checked = preferencesStore.typewriterMode;
      break;

    case 'view.toggleFocusMode':
      state.checked = preferencesStore.focusMode;
      break;

    // 格式菜单
    case 'format.bold':
    case 'format.italic':
    case 'format.underline':
      state.enabled = editorStore.hasSelection;
      break;

    // 窗口菜单
    case 'window.switchWindow':
    case 'window.showAllWindows':
      state.enabled = windowManager.windows.size > 1;
      break;

    default:
      break;
  }

  return state;
}
```

### 4.2 渲染进程菜单处理

```typescript
// src/renderer/composables/useMenuHandler.ts
import { onMounted, onUnmounted } from 'vue';
import { ipcRenderer } from 'electron';
import { useTabsStore } from '@/stores/tabs';
import { useEditorStore } from '@/stores/editor';

export function useMenuHandler() {
  const tabsStore = useTabsStore();
  const editorStore = useEditorStore();

  const menuHandlers = {
    // 文件菜单
    'menu:newFile': () => {
      tabsStore.createNewTab();
    },

    'menu:openFile': (filePath: string) => {
      tabsStore.openFile(filePath);
    },

    'menu:openFolder': (folderPath: string) => {
      // 打开文件夹到文件树
      editorStore.openFolder(folderPath);
    },

    'menu:save': () => {
      const activeTab = tabsStore.activeTab;
      if (activeTab) {
        tabsStore.saveTab(activeTab.id);
      }
    },

    'menu:saveAs': () => {
      const activeTab = tabsStore.activeTab;
      if (activeTab) {
        tabsStore.saveTabAs(activeTab.id);
      }
    },

    'menu:closeTab': () => {
      if (tabsStore.activeTabId) {
        tabsStore.removeTab(tabsStore.activeTabId);
      }
    },

    // 编辑菜单
    'menu:undo': () => {
      editorStore.undo();
    },

    'menu:redo': () => {
      editorStore.redo();
    },

    'menu:cut': () => {
      document.execCommand('cut');
    },

    'menu:copy': () => {
      document.execCommand('copy');
    },

    'menu:paste': () => {
      document.execCommand('paste');
    },

    // 视图菜单
    'menu:toggleSidebar': () => {
      editorStore.toggleSidebar();
    }
  };

  onMounted(() => {
    // 注册所有菜单事件监听
    Object.entries(menuHandlers).forEach(([channel, handler]) => {
      ipcRenderer.on(channel, handler as any);
    });
  });

  onUnmounted(() => {
    // 清理所有监听
    Object.entries(menuHandlers).forEach(([channel, handler]) => {
      ipcRenderer.removeListener(channel, handler as any);
    });
  });
}
```
