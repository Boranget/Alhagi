### P4.3 统一 Handler 内部模式

所有 Handler 统一采用构造函数注入 + execute 方法：

```typescript
export class SaveFileHandler implements ICommandHandler {
  constructor(
    private tabService: ITabService,
    private tabRepository: ITabRepository
  ) {}

  async execute(): Promise<CommandResult> {
    const activeTabId = this.tabRepository.getActiveTabId()
    if (!activeTabId) return { success: false, error: 'No active tab' }
    const tab = this.tabRepository.getById(activeTabId)
    if (!tab?.filePath) return { success: false, error: 'No file path' }
    await this.tabService.saveFile(tab.filePath, tab.content)
    return { success: true }
  }
}
```
